// Function to check Redux store state
function checkReduxState() {
  console.log('=== REDUX STATE DEBUG ===');
  
  if (typeof window !== 'undefined' && window.store) {
    const state = window.store.getState();
    console.log('Full Redux State:', state);
    console.log('Auth State:', state.auth);
    console.log('Token:', state.auth?.token);
    console.log('Refresh Token:', state.auth?.refreshToken);
    console.log('Is Authenticated:', state.auth?.isAuthenticated);
    console.log('User:', state.auth?.user);
    console.log('Loading:', state.auth?.isLoading);
    console.log('Error:', state.auth?.error);
  } else {
    console.log('Redux store not available in window object');
  }
}
 
// Function to check localStorage
function checkLocalStorage() {
  console.log('=== LOCAL STORAGE DEBUG ===');
  
  // Check Redux Persist data
  const persistedState = localStorage.getItem('persist:root');
  if (persistedState) {
    try {
      const parsed = JSON.parse(persistedState);
      console.log('Persisted State:', parsed);
      
      if (parsed.auth) {
        const authState = JSON.parse(parsed.auth);
        console.log('Persisted Auth State:', authState);
        console.log('Persisted Token:', authState.token);
        console.log('Persisted Refresh Token:', authState.refreshToken);
        console.log('Persisted Is Authenticated:', authState.isAuthenticated);
        console.log('Persisted User:', authState.user);
      }
    } catch (error) {
      console.error('Error parsing persisted state:', error);
    }
  } else {
    console.log('No persisted state found in localStorage');
  }
  
  // Check individual auth keys
  const authToken = localStorage.getItem('auth_token');
  const refreshToken = localStorage.getItem('refresh_token');
  const userData = localStorage.getItem('user_data');
  
  console.log('Individual localStorage items:');
  console.log('auth_token:', authToken);
  console.log('refresh_token:', refreshToken);
  console.log('user_data:', userData);
}

// Function to test API call with current token
async function testAPICall() {
  console.log('=== API CALL TEST ===');
  
  const token = getTokenFromState();
  console.log('Using token:', token);
  
  if (!token) {
    console.log('No token available for API call');
    return;
  }
  
  try {
    const response = await fetch('http://127.0.0.1:8000/api/auth/users/', {
      method: 'GET',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${token}`
      }
    });
    
    console.log('API Response Status:', response.status);
    console.log('API Response Headers:', Object.fromEntries(response.headers.entries()));
    
    if (response.ok) {
      const data = await response.json();
      console.log('API Response Data:', data);
    } else {
      const errorText = await response.text();
      console.log('API Error Response:', errorText);
    }
  } catch (error) {
    console.error('API Call Error:', error);
  }
}
 
// Helper function to get token from current state
function getTokenFromState() {
  if (typeof window !== 'undefined' && window.store) {
    const state = window.store.getState();
    return state.auth?.token;
  }
  
  // Fallback to localStorage
  const persistedState = localStorage.getItem('persist:root');
  if (persistedState) {
    try {
      const parsed = JSON.parse(persistedState);
      if (parsed.auth) {
        const authState = JSON.parse(parsed.auth);
        return authState.token;
      }
    } catch (error) {
      console.error('Error getting token from localStorage:', error);
    }
  }
  
  return null;
}

// Function to check if user needs to login
function checkLoginStatus() {
  console.log('=== LOGIN STATUS CHECK ===');
  
  const token = getTokenFromState();
  const isAuthenticated = window.store?.getState()?.auth?.isAuthenticated;
  
  console.log('Has token:', !!token);
  console.log('Is authenticated:', isAuthenticated);
  
  if (!token || !isAuthenticated) {
    console.log('❌ User needs to login');
    console.log('Navigate to: http://localhost:3000/login');
  } else {
    console.log('✅ User appears to be authenticated');
  }
}
